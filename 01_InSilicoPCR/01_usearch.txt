# --------------------------------------------------------- #
# Set working directory
# --------------------------------------------------------- #
cd WORKING_FOLDER
output_dir=01_usearchOut
mkdir ${output_dir}
mkdir ${output_dir}/1_EachPrimerSet
mkdir ${output_dir}/2_EachPrimer


# --------------------------------------------------------- #
# in silico PCR
# --------------------------------------------------------- #

# Analyse each primer set using usearch -search_pcr
## maxdiffs = 1-3 for each primer set

while read primer_set
do
  mkdir ${output_dir}/${primer_set}

  usearch -search_pcr data_dbseq/vertebrate_db.fa \
  -db data_primerset/each_primerset/${primer_set}.fa \
  -maxdiffs 3 -strand both -minamp 30 -maxamp 2000 -threads 36 \
  -pcrout ${output_dir}/1_EachPrimerSet/${primer_set}/hits_all_text.txt \
  -ampout ${output_dir}/1_EachPrimerSet/${primer_set}/hits_all.fa

done < data_primerset/primer_names.txt


# Analyze each primer using usearch -search_oligodb
## maxdiffs = 4

while read primer
do
  mkdir ${output_dir}/${primer}

  usearch -search_oligodb data_dbseq/vertebrate_db.fa \
  -db data_primerset/each_primer/${primer}.fa \
  -maxdiffs 4 -strand both -threads 36 \
  -userout ${output_dir}/2_EachPrimer/${primer}/hits_all_text.txt \
  -userfields query+target+qstrand+diffs+tlo+thi+trowdots

done < data_primerset/primer_names_each.txt


# --------------------------------------------------------- #
# in silico PCR for intraspecific variation
# --------------------------------------------------------- #
# Create another output directory
mkdir ${output_dir}/1_EachPrimerSet/CheckEditDistance_Cetacea
mkdir ${output_dir}/1_EachPrimerSet/CheckEditDistance_MarineMammal


# For Cetacea species
while read primer_set
do
  mkdir ${output_dir}/1_EachPrimerSet/CheckEditDistance_Cetacea/${primer_set}

  usearch -search_pcr data_dbseq/mt_refseq_cetacea.fa \
  -db data_primerset/each_primerset/${primer_set}.fa \
  -maxdiffs 3 -strand both -minamp 30 -maxamp 2000 -threads 36 \
  -pcrout ${output_dir}/1_EachPrimerSet/CheckEditDistance_Cetacea/${primer_set}/hits_all_text.txt \
  -ampout ${output_dir}/1_EachPrimerSet/CheckEditDistance_Cetacea/${primer_set}/hits_all.fa

done < data_primerset/primer_names.txt


# For Other mammal species
while read primer_set

  usearch -search_pcr data_dbseq/mt_refseq_other_marine_mammals.fa \
  -db data_primerset/each_primerset/${primer_set}.fa \
  -maxdiffs 3 -strand both -minamp 30 -maxamp 2000 -threads 36 \
  -pcrout ${output_dir}/1_EachPrimerSet/CheckEditDistance_MarineMammal/${primer_set}/hits_all_text.txt \
  -ampout ${output_dir}/1_EachPrimerSet/CheckEditDistance_MarineMammal/${primer_set}/hits_all.fa

done < data_primerset/primer_names.txt
